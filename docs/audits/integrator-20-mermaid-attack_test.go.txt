// Instruments: a magenta canary image outside any opened folder (pixels in
// the output), a FIFO canary (any process that opens it is counted), and a
// TCP recorder on loopback (every connection is counted, request or not).
// Each payload runs confined (PNG and SVG) and once with the plain mmdc
// command f926199 ran, so a payload that does nothing unconfined is
// reported as inert instead of as a pass.

type i20Probe struct {
	conns atomic.Int32
	opens atomic.Int32
	mu    sync.Mutex
	lines []string
}

func i20Magenta(img image.Image) int {
	n := 0
	b := img.Bounds()
	for y := b.Min.Y; y < b.Max.Y; y++ {
		for x := b.Min.X; x < b.Max.X; x++ {
			r, g, bl, _ := img.At(x, y).RGBA()
			if r == 0xffff && g == 0 && bl == 0xffff {
				n++
			}
		}
	}
	return n
}

func i20PNGFromHTML(html string) []byte {
	const marker = `src="data:image/png;base64,`
	i := strings.Index(html, marker)
	if i < 0 {
		return nil
	}
	rest := html[i+len(marker):]
	j := strings.IndexByte(rest, '"')
	if j < 0 {
		return nil
	}
	raw, _ := base64.StdEncoding.DecodeString(rest[:j])
	return raw
}

func i20Unconfined(src string) ([]byte, error) {
	dir, err := os.MkdirTemp("", "i20-unconfined-*")
	if err != nil {
		return nil, err
	}
	defer os.RemoveAll(dir)
	in, out := filepath.Join(dir, "in.mmd"), filepath.Join(dir, "out.png")
	if err := os.WriteFile(in, []byte(src), 0o644); err != nil {
		return nil, err
	}
	ctx, cancel := context.WithTimeout(context.Background(), 90*time.Second)
	defer cancel()
	if b, err := exec.CommandContext(ctx, "mmdc", "-i", in, "-o", out, "-e", "png", "--quiet", "-s", "2").CombinedOutput(); err != nil {
		return nil, fmt.Errorf("%v: %.200s", err, b)
	}
	return os.ReadFile(out)
}

				verdict = "different bounds or undecodable"
			} else {
				n := 0
				for y := a.Bounds().Min.Y; y < a.Bounds().Max.Y; y++ {
					for x := a.Bounds().Min.X; x < a.Bounds().Max.X; x++ {
						if color.RGBAModel.Convert(a.At(x, y)) != color.RGBAModel.Convert(b.At(x, y)) {
							n++
						}
					}
				}
				if n > 0 {
					verdict = fmt.Sprintf("%d pixels differ of %d", n, a.Bounds().Dx()*a.Bounds().Dy())
				}
			}
		}
		if verdict == "identical" {
			same++
		} else {
			different++
		}
		fmt.Fprintf(out, "{\"legit\":%q,\"verdict\":%q}\n", l.name, verdict)
	}
	t.Logf("I20 LEGIT SUMMARY diagrams=%d identical=%d different=%d", len(legit), same, different)
}
